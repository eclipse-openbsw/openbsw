231  {
232  auto const f = typename TL::head();
233  auto const& cmpType
234  = *static_cast<internal::column<typename TL::head::value_type, Table>*>(&table.columns);
235  auto const aType = ParamConvert<typename TL::head::value_type>::to_param_type(cmpType[a]);
236  auto const bType = ParamConvert<typename TL::head::value_type>::to_param_type(cmpType[b]);
237 
238  if (f(bType, aType))
239  {
240  return true;
241  }
242  if (f(aType, bType))
243  {
244  return false;
245  }
246  return recurse::less_than(table, a, b);

409  template<typename Cmp, typename Table, typename Q, typename It>
410  static void move_while(Table& table, Q& q, size_t const dst, It& it)
411  {
412  using tr = type_list_recurse<Cmp>;
413  using OrdCol = typename ordering_for<Cmp>::type;
414 
415  static_assert(
416  ::std::is_base_of<OrdCol, typename Table::column_data>::value,
417  "ordered operation requires the schema to declare a matching "
418  "shed::ordering<Cmp...> column, listing the comparators in the same order as the "
419  "ordered<Cmp...> operation");
420 
421  auto& ml = static_cast<typename Table::state_data*>(&table.columns)->ml;
422  auto& ord = *static_cast<OrdCol*>(&table.columns);
423 
424  auto csfr = [&q, &table](size_t const i) -> move_op
425  { return call_system_func_ret<::shed::move_op, Q, Table, size_t, Args...>(q, table, i); };
426 
427  auto pred = [&table](size_t const a, size_t const b) -> bool
428  { return tr::less_than(table, a, b); };
429 
430  auto const begin = ord._scratch.begin();
431  auto end = begin;
432  auto f = [&table, &end](size_t const i) -> bool
433  {
434  if (tr::all_set(table, i))
435  {
436  *end = static_cast<internal::multi_list::idx_type>(i);
437  ++end;
438  }
439  return true;
440  };
441  it.iter(f);
442 
443  ::shed::internal::move_while_impl(begin, end, ml, dst, pred, csfr);
444  }

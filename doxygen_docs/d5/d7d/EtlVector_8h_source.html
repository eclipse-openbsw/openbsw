125 template<class T, class LENGTH_TYPE>
126 void DynamicArrayParser<T, LENGTH_TYPE>::parse(
127  ::etl::ivector<T>& obj, SomeIpParser& parser, size_t const minLength, bool const isBigEndian)
128 {
129  obj.clear();
130 
131  parser.bigEndian();
132 
133  LENGTH_TYPE const numBytes
134  = static_cast<LENGTH_TYPE>(LengthParserHelper<LENGTH_TYPE>::parseLength(parser));
135 
136  if (!parser.isGood())
137  {
138  return;
139  }
140 
141  if (isBigEndian == false)
142  {
143  parser.littleEndian();
144  }
145 
146  size_t pos = 0U;
147  size_t const endPosition = parser.getCurrentPosition() + static_cast<size_t>(numBytes);
148  while (parser.getCurrentPosition() < endPosition)
149  {
150  if (pos >= obj.max_size())
151  {
152  // fatal. Trying to read in too many values!
153  obj.clear();
154  parser.setFailure();
155  return;
156  }
157 
158  T element{};
159  parser >> element;
160  if (!parser.isGood())
161  {
162  obj.clear();
163  return;
164  }
165  obj.push_back(element);
166 
167  pos++;
168  }
169 
170  if (pos < minLength)
171  {
172  parser.setFailure();
173  obj.clear();
174  }
175 }

22 // multi_list is a data structure to maintain membership information for N elements in M sets.
23 // The items are always the numbers 0..N-1, intended to be used as indices into another data
24 // structure on the side (in case of shed, multiple other arrays)
25 //
26 // Invariants:
27 // - The total number of items and also sets/buckets stays constant
28 // - Every index is a member of exactly one set at all times
29 //
30 // Characteristics:
31 // - Memory consumption is O(N + M)
32 // - Iteration of members of any set is O(Nx), with Nx being the number of elements in the given
33 // set
34 // - Iteration of members not in a set is O(N)
35 // - Transferring of any number of elements from set A to set B is O(Na)
36 // - Operations on empty sets are always O(1)
37 // - All operations use sequential memory access patterns with high locality
38 // - All operations are done in place and don't require extra memory
39 //
40 // Implementation:
41 // The idea is to use M linked lists interleaved inside one shared array.
42 // By using indices instead of pointers a smaller type (uint16_t) than pointer size can be used
43 // to save memory and increase locality.
44 // To make set transfers faster, the lists are doubly-linked, with the backwards links stored
45 // in a separate array on the side to make forward iteration faster by not polluting the cpu cache
46 // with backward links not needed for forward iteration.
47 // The index of any element serves as it's address, but also as it's associated value/"payload"
48 // Therefore no extra memory for "payload" is required.

54 struct multi_list
55 {
56  using idx_type = uint16_t;
57 
58  static constexpr bool is_size_valid(size_t const n, size_t const buckets)
59  {
60  return (n + buckets) <= std::numeric_limits<idx_type>::max();
61  }
62 
63  static constexpr size_t memory_for(size_t const n, size_t const buckets)
64  {
65  // (n + buckets) has to be <= maxint(idx_type), in order for multi_list to work;
66  // because of a limited support for constexpr in diab we can't have an extra condition
67  // in this function
68  return sizeof(multi_list) + (2 * sizeof(idx_type) * (n + buckets));
69  }

81  template<typename F>
82  void iter(F const f) const
83  {
84  for (size_t pos = items[bucket_n]; pos != bucket_n; pos = items[pos])
85  {
86  if (!f(pos))
87  {
88  break;
89  }
90  }

104  template<typename F>
105  void iter(F&& f) const
106  {
107  size_t const n = self->_n;
108  idx_type const* const items = self->items();
109  size_t pos = items[n + bucket];
110  size_t i = 0;
111 
112  do
113  {
114  ++i;
115 
116  // This loop skips slots that are in the bucket
117  while ((n - i) == pos)
118  {
119  ++i;
120  pos = items[pos];
121  }
122 
123  } while ((i < n + 1) && std::forward<F>(f)(n - i));
124  }
125  };

149  template<typename F>
150  size_t move_if(size_t src, size_t dst, F&& f)
151  {
152  src = src + _n;
153  dst = dst + _n;
154  size_t moved = 0;
155 
156  while (items()[src] < _n)
157  {
158  auto const s = items()[src];
159  auto const r = std::forward<F>(f)(s);
160  if ((r == move_op::MOVE) || (r == move_op::MOVE_DONE))
161  {
162  dst = move_node(s, dst);
163  ++moved;
164  }
165  else
166  {
167  src = items()[src];
168  }
169  if ((r == move_op::SKIP_DONE) || (r == move_op::MOVE_DONE))
170  {
171  break;
172  }
173  }
174  return moved;
175  }

168 template<class DataLinkLayer>
169 void DoCanRangeExtendedAddressingFilter<DataLinkLayer>::init(
170  DataLinkAddressType const baseCanId,
171  uint16_t const baseTransportId,
172  uint16_t const count,
173  FunctionalAddressSliceType const functionalAddresses,
174  FrameCodecType const& codec)
175 {
176  ETL_ASSERT(count != 0U, ETL_ERROR_GENERIC("range must not be empty"));
177  ETL_ASSERT(
178  (baseCanId + (count - 1U)) <= ::can::CanId::MAX_RAW_BASE_ID,
179  ETL_ERROR_GENERIC("range must fit into a base (11 bit) CAN identifier"));
180  ETL_ASSERT(
181  (static_cast<uint32_t>(baseTransportId) + (count - 1U)) <= 0xffU,
182  ETL_ERROR_GENERIC("range must fit into an address extension byte"));
183 
184  _baseCanId = baseCanId;
185  _baseTransportId = baseTransportId;
186  _count = count;
187  _codec = &codec;
188  _functionalAddresses = functionalAddresses;
189  IntervalFilter::add(baseCanId, baseCanId + (count - 1U));
190 }

192 template<class DataLinkLayer>
193 typename DoCanRangeExtendedAddressingFilter<DataLinkLayer>::FrameCodecType const*
194 DoCanRangeExtendedAddressingFilter<DataLinkLayer>::getTransmissionParameters(
195  DoCanTransportAddressPair const& transportAddressPair,
196  DataLinkAddressPairType& dataLinkAddressPair) const
197 {
198  DataLinkAddressType sourceCanId;
199  DataLinkAddressType targetCanId;
200  if (!transportIdToCanId(transportAddressPair.getSourceId(), sourceCanId)
201  || !transportIdToCanId(transportAddressPair.getTargetId(), targetCanId))
202  {
203  return nullptr;
204  }
205 
206  dataLinkAddressPair = DataLinkAddressPairType(
207  AddressingType::pack(targetCanId, transportAddressPair.getSourceId()),
208  AddressingType::pack(sourceCanId, transportAddressPair.getTargetId()));
209  return _codec;
210 }

212 template<class DataLinkLayer>
213 typename DoCanRangeExtendedAddressingFilter<DataLinkLayer>::FrameCodecType const*
214 DoCanRangeExtendedAddressingFilter<DataLinkLayer>::getReceptionParameters(
215  DataLinkAddressType receptionAddress,
216  DoCanTransportAddressPair& transportAddressPair,
217  DataLinkAddressType& transmissionAddress) const
218 {
219  DataLinkAddressType const canId = AddressingType::canIdOf(receptionAddress);
220  // the address extension byte identifies the target of the received frame, i.e. this node.
221  uint16_t const targetId = AddressingType::addressExtensionOf(receptionAddress);
222 
223  uint16_t sourceId;
224  if (!canIdToTransportId(canId, sourceId))
225  {
226  return nullptr;
227  }
228 
229  DataLinkAddressType targetCanId;
230  if (!transportIdToCanId(targetId, targetCanId))
231  {
232  return nullptr;
233  }
234 
235  transportAddressPair = DoCanTransportAddressPair(sourceId, targetId);
236  // ISO 15765-2 forbids multi-frame requests to a functional (broadcast) target, so such a
237  // request deliberately reports an invalid transmission address rather than a real one:
238  // DoCanReceiver rejects any multi-frame request whose transmission address is invalid, while
239  // single-frame functional requests remain unaffected, since the actual response is always
240  // addressed independently, once the request has been fully received.
241  transmissionAddress = isFunctionalAddress(targetId)
242  ? DataLinkLayerType::INVALID_ADDRESS
243  : AddressingType::pack(targetCanId, sourceId);
244  return _codec;
245 }

266 template<class DataLinkLayer>
267 bool DoCanRangeExtendedAddressingFilter<DataLinkLayer>::transportIdToCanId(
268  uint16_t const transportId, DataLinkAddressType& canId) const
269 {
270  if ((transportId < _baseTransportId) || (transportId >= (_baseTransportId + _count)))
271  {
272  return false;
273  }
274  canId = _baseCanId + (transportId - _baseTransportId);
275  return true;
276 }

278 template<class DataLinkLayer>
279 bool DoCanRangeExtendedAddressingFilter<DataLinkLayer>::canIdToTransportId(
280  DataLinkAddressType const canId, uint16_t& transportId) const
281 {
282  if ((canId < _baseCanId) || (canId >= (_baseCanId + _count)))
283  {
284  return false;
285  }
286  transportId = static_cast<uint16_t>(_baseTransportId + (canId - _baseCanId));
287  return true;
288 }

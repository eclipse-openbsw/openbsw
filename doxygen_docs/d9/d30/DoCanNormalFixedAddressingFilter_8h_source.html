162 template<class DataLinkLayer>
163 void DoCanNormalFixedAddressingFilter<DataLinkLayer>::init(
164  FunctionalAddressSliceType const functionalAddresses,
165  TesterAddressSliceType const allowedTesters,
166  FrameCodecType const& codec)
167 {
168  _functionalAddresses = functionalAddresses;
169  _allowedTesters = allowedTesters;
170  _codec = &codec;
171 
172  uint32_t const physicalBase
173  = AddressingType::pack(AddressingType::PHYSICAL_ADDRESSING_ID_BASE, 0x00U, 0x00U);
174  uint32_t const functionalBase
175  = AddressingType::pack(AddressingType::FUNCTIONAL_ADDRESSING_ID_BASE, 0x00U, 0x00U);
176  // target/source address bytes (bits 0-15) may take any value.
177  MaskFilter::add(physicalBase, physicalBase | 0xFFFFU);
178  MaskFilter::add(functionalBase, functionalBase | 0xFFFFU);
179 }

181 template<class DataLinkLayer>
182 typename DoCanNormalFixedAddressingFilter<DataLinkLayer>::FrameCodecType const*
183 DoCanNormalFixedAddressingFilter<DataLinkLayer>::getTransmissionParameters(
184  DoCanTransportAddressPair const& transportAddressPair,
185  DataLinkAddressPairType& dataLinkAddressPair) const
186 {
187  uint8_t sourceAddress;
188  uint8_t targetAddress;
189  if (!toAddressByte(transportAddressPair.getSourceId(), sourceAddress)
190  || !toAddressByte(transportAddressPair.getTargetId(), targetAddress))
191  {
192  return nullptr;
193  }
194 
195  // transmission is always physical, functional requests are single-frame only and therefore
196  // never require this connection-oriented (multi-frame/flow-control) path.
197  dataLinkAddressPair = DataLinkAddressPairType(
198  AddressingType::pack(
199  AddressingType::PHYSICAL_ADDRESSING_ID_BASE, sourceAddress, targetAddress),
200  AddressingType::pack(
201  AddressingType::PHYSICAL_ADDRESSING_ID_BASE, targetAddress, sourceAddress));
202  return _codec;
203 }

205 template<class DataLinkLayer>
206 typename DoCanNormalFixedAddressingFilter<DataLinkLayer>::FrameCodecType const*
207 DoCanNormalFixedAddressingFilter<DataLinkLayer>::getReceptionParameters(
208  DataLinkAddressType const receptionAddress,
209  DoCanTransportAddressPair& transportAddressPair,
210  DataLinkAddressType& transmissionAddress) const
211 {
212  uint32_t const addressingIdBase = AddressingType::addressingIdBaseOf(receptionAddress);
213  uint8_t const targetAddress = AddressingType::targetAddressOf(receptionAddress);
214  uint8_t const sourceAddress = AddressingType::sourceAddressOf(receptionAddress);
215  bool isFunctional = false;
216 
217  // Reject requests from testers that are not configured.
218  // Returning nullptr makes the request silently disappear
219  // before UDS processing starts.
220  if (!_allowedTesters.empty())
221  {
222  if (!isAllowedTester(sourceAddress))
223  {
224  return nullptr;
225  }
226  }
227 
228  if (addressingIdBase == AddressingType::PHYSICAL_ADDRESSING_ID_BASE)
229  {
230  if (isFunctionalAddress(targetAddress))
231  {
232  // the target address byte is reserved for functional addressing in this
233  // configuration.
234  return nullptr;
235  }
236  }
237  else if (addressingIdBase == AddressingType::FUNCTIONAL_ADDRESSING_ID_BASE)
238  {
239  if (!isFunctionalAddress(targetAddress))
240  {
241  return nullptr;
242  }
243  isFunctional = true;
244  }
245  else
246  {
247  return nullptr;
248  }
249 
250  // the reported target address is always the raw target address byte taken from the wire,
251  // unmodified; it is up to the message receiver to recognize it as a functional (broadcast)
252  // address if it needs to.
253  transportAddressPair = DoCanTransportAddressPair(sourceAddress, targetAddress);
254  // ISO 15765-2 forbids multi-frame requests to a functional (broadcast) target, so a
255  // functional request deliberately reports an invalid transmission address rather than a
256  // real one: DoCanReceiver rejects any multi-frame request whose transmission address is
257  // invalid, while single-frame functional requests remain unaffected, since the actual
258  // response is always addressed independently, once the request has been fully received.
259  transmissionAddress
260  = isFunctional
261  ? DataLinkLayerType::INVALID_ADDRESS
262  : AddressingType::pack(
263  AddressingType::PHYSICAL_ADDRESSING_ID_BASE, sourceAddress, targetAddress);
264  return _codec;
265 }

296  uint16_t const transportId, uint8_t& address)
297 {
298  if (transportId > 0xFFU)

300  return false;
301  }

180 template<typename MT, uint8_t RL>
181 void ProxyMethod<MT, RL>::answerRequest(core::Message const& msg)
182 {
183  core::Future* const future = futureMatchingRequestId(msg);
184  if (future != nullptr)
185  {
186  if (future->state != core::Future::State::Ready)
187  {
188  notifyWithError(*future);
189  }
190  else
191  {
192  notifyWithResult(msg, *future);
193  }
194  }
195 }

197 template<typename MT, uint8_t RL>
198 template<typename T>
199 etl::enable_if_t<!etl::is_same_v<T, void>, etl::expected<uint16_t, core::HRESULT>>
200 ProxyMethod<MT, RL>::callMethod(
201  core::ProxyBase const& proxy, T const& input, uint16_t const methodId, Callback const cbk)
202 {
203  if (!proxy.isInitialized())
204  {
205  return etl::unexpected<core::HRESULT>(etl::in_place, core::HRESULT::NotRegistered);
206  }
207 
208  core::Future* future = Base::obtainFuture();
209  if (future == nullptr)
210  {
211  return etl::unexpected<core::HRESULT>(etl::in_place, core::HRESULT::RequestPoolDepleted);
212  }
213 
214  core::Message msg = proxy.generateMessageHeader(methodId, future->requestId);
215  auto const index = etl::distance(_futuresArray.begin(), future);
216  _callbacks[index] = cbk;
217  core::HRESULT ret = core::MessagePayloadBuilder::getInstance().allocate(input, msg);
218  return Base::callMethod(proxy, msg, ret);
219 }

244 template<typename MT, uint8_t RL>
245 void ProxyMethod<MT, RL>::notifyWithError(core::Future& future)
246 {
247  auto const index = etl::distance(_futuresArray.begin(), &future);
248  _callbacks[index].call_if(
249  etl::unexpected<core::Future::State>(etl::in_place_t{}, future.state));
250  clearFutureAndCallback(future, _callbacks[index]);
251 }
252 
253 template<typename MT, uint8_t RL>
254 void ProxyMethod<MT, RL>::notifyWithResult(core::Message const& msg, core::Future& future)
255 {
256  auto const index = etl::distance(_futuresArray.begin(), &future);
257  if constexpr (etl::is_same_v<OutputType, void>)
258  {
259  _callbacks[index].call_if(Result());
260  }
261  else
262  {
263  _callbacks[index].call_if(Result(
264  etl::in_place_t{},
265  etl::cref(core::MessagePayloadBuilder::getInstance().readPayload<OutputType>(msg))));
266  }
267  clearFutureAndCallback(future, _callbacks[index]);
268 }

270 template<typename MT, uint8_t RL>
271 void ProxyMethod<MT, RL>::clearFutureAndCallback(core::Future& future, Callback& callback)
272 {
273  future.callerTimestamp = 0U;
274  future.requestId = core::INVALID_REQUEST_ID;
275  future.state = core::Future::State::Invalid;
276  callback = {};
277 }

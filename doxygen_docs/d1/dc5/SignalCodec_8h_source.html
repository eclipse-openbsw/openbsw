52 template<size_t BitLength, typename Value>
53 void writeSignal(etl::bit_stream_writer& stream, Value const& value)
54 {
55  static_assert(BitLength > 0U, "Signal width must be positive.");
56  if constexpr (detail::IsArray<Value>::value)
57  {
58  using Element = typename Value::value_type;
59  constexpr size_t elementBits = etl::numeric_limits<Element>::digits;
60  static_assert(
61  BitLength == Value::SIZE * elementBits, "Array width must cover all elements.");
62  for (auto const& element : value)
63  {
64  writeSignal<elementBits>(stream, element);
65  }
66  }
67  else if constexpr (etl::is_enum<Value>::value)
68  {
69  using Underlying = typename etl::underlying_type<Value>::type;
70  writeSignal<BitLength>(stream, static_cast<Underlying>(value));
71  }
72  else if constexpr (detail::HasRawValue<Value>::value)
73  {
74  writeSignal<BitLength>(stream, value.value());
75  }
76  else if constexpr (etl::is_same<Value, bool>::value)
77  {
78  static_assert(BitLength == 1U, "Boolean signals occupy one bit.");
79  stream.write_unchecked(value);
80  }
81  else
82  {
83  static_assert(
84  etl::is_integral<Value>::value && etl::is_unsigned<Value>::value,
85  "Scalar signals must use unsigned integral types.");
86  static_assert(
87  BitLength <= etl::numeric_limits<Value>::digits,
88  "Signal width exceeds its storage type.");
89  stream.write_unchecked(value, BitLength);
90  }
91 }

105 template<typename Value, size_t BitLength>
106 Value readSignal(etl::bit_stream_reader& stream)
107 {
108  static_assert(BitLength > 0U, "Signal width must be positive.");
109  if constexpr (detail::IsArray<Value>::value)
110  {
111  using Element = typename Value::value_type;
112  constexpr size_t elementBits = etl::numeric_limits<Element>::digits;
113  static_assert(
114  BitLength == Value::SIZE * elementBits, "Array width must cover all elements.");
115  Value result{};
116  for (auto& element : result)
117  {
118  element = readSignal<Element, elementBits>(stream);
119  }
120  return result;
121  }
122  else if constexpr (etl::is_enum<Value>::value)
123  {
124  using Underlying = typename etl::underlying_type<Value>::type;
125  return static_cast<Value>(readSignal<Underlying, BitLength>(stream));
126  }
127  else if constexpr (detail::HasRawValue<Value>::value)
128  {
129  using Underlying = decltype(etl::declval<Value const&>().value());
130  return Value(readSignal<Underlying, BitLength>(stream));
131  }
132  else if constexpr (etl::is_same<Value, bool>::value)
133  {
134  static_assert(BitLength == 1U, "Boolean signals occupy one bit.");
135  return stream.read_unchecked<bool>();
136  }
137  else
138  {
139  static_assert(
140  etl::is_integral<Value>::value && etl::is_unsigned<Value>::value,
141  "Scalar signals must use unsigned integral types.");
142  static_assert(
143  BitLength <= etl::numeric_limits<Value>::digits,
144  "Signal width exceeds its storage type.");
145  return stream.read_unchecked<Value>(BitLength);
146  }
147 }

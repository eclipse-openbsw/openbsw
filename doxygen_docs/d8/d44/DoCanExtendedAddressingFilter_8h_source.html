185 template<class DataLinkLayer, class AddressEntry>
186 void DoCanExtendedAddressingFilter<DataLinkLayer, AddressEntry>::init(
187  AddressEntrySliceType const addressEntries,
188  FunctionalAddressSliceType const functionalAddresses,
189  FrameCodecType const& codec)
190 {
191  // Address entries must be non-zero in size
192  ETL_ASSERT(
193  addressEntries.size() != 0, ETL_ERROR_GENERIC("list of addresses must not be empty"));
194 
195  _entries = addressEntries;
196  _codec = &codec;
197  _functionalAddresses = functionalAddresses;
198  DataLinkAddressType prevCanId = 0U;
199  typename AddressEntrySliceType::const_iterator it = _entries.begin();
200 
201  for (; it != _entries.end(); ++it)
202  {
203  DataLinkAddressType const canId = it->_canId;
204  if (it != _entries.begin())
205  {
206  ETL_ASSERT(
207  prevCanId <= canId,
208  ETL_ERROR_GENERIC("can id must be ascending (duplicates allowed)"));
209  }
210  prevCanId = canId;
211  BitFieldFilter::add(canId);
212  }
213 }

215 template<class DataLinkLayer, class AddressEntry>
216 typename DoCanExtendedAddressingFilter<DataLinkLayer, AddressEntry>::FrameCodecType const*
217 DoCanExtendedAddressingFilter<DataLinkLayer, AddressEntry>::getTransmissionParameters(
218  DoCanTransportAddressPair const& transportAddressPair,
219  DataLinkAddressPairType& dataLinkAddressPair) const
220 {
221  AddressEntryType const* const sourceEntry
222  = findEntryByTransportId(transportAddressPair.getSourceId());
223  AddressEntryType const* const targetEntry
224  = findEntryByTransportId(transportAddressPair.getTargetId());
225  if ((sourceEntry == nullptr) || (targetEntry == nullptr))
226  {
227  return nullptr;
228  }
229 
230  dataLinkAddressPair = DataLinkAddressPairType(
231  AddressingType::pack(targetEntry->_canId, transportAddressPair.getSourceId()),
232  AddressingType::pack(sourceEntry->_canId, transportAddressPair.getTargetId()));
233  return _codec;
234 }

236 template<class DataLinkLayer, class AddressEntry>
237 typename DoCanExtendedAddressingFilter<DataLinkLayer, AddressEntry>::FrameCodecType const*
238 DoCanExtendedAddressingFilter<DataLinkLayer, AddressEntry>::getReceptionParameters(
239  DataLinkAddressType receptionAddress,
240  DoCanTransportAddressPair& transportAddressPair,
241  DataLinkAddressType& transmissionAddress) const
242 {
243  uint32_t const canId = AddressingType::canIdOf(receptionAddress);
244  // the address extension byte identifies the target of the received frame, i.e. this node.
245  uint16_t const targetId = AddressingType::addressExtensionOf(receptionAddress);
246 
247  AddressEntryType const* const sourceEntry
248  = findEntryByCanId(canId, _entries.begin(), _entries.end());
249  if (sourceEntry == nullptr)
250  {
251  return nullptr;
252  }
253 
254  AddressEntryType const* const targetEntry = findEntryByTransportId(targetId);
255  if (targetEntry == nullptr)
256  {
257  return nullptr;
258  }
259 
260  transportAddressPair = DoCanTransportAddressPair(sourceEntry->_transportId, targetId);
261  // ISO 15765-2 forbids multi-frame requests to a functional (broadcast) target, so such a
262  // request deliberately reports an invalid transmission address rather than a real one:
263  // DoCanReceiver rejects any multi-frame request whose transmission address is invalid, while
264  // single-frame functional requests remain unaffected, since the actual response is always
265  // addressed independently, once the request has been fully received.
266  transmissionAddress
267  = isFunctionalAddress(targetId)
268  ? DataLinkLayerType::INVALID_ADDRESS
269  : AddressingType::pack(targetEntry->_canId, sourceEntry->_transportId);
270  return _codec;
271 }

294 typename DoCanExtendedAddressingFilter<DataLinkLayer, AddressEntry>::AddressEntryType const*
295 DoCanExtendedAddressingFilter<DataLinkLayer, AddressEntry>::findEntryByCanId(
296  DataLinkAddressType const canId,
297  typename AddressEntrySliceType::const_iterator const beginIt,
298  typename AddressEntrySliceType::const_iterator const endIt)
299 {
300  AddressEntryType key{};
301  key._canId = canId;
302  typename AddressEntrySliceType::const_iterator const it
303  = ::etl::lower_bound(beginIt, endIt, key, &lessCanId);
304  return ((it != endIt) && (it->_canId == canId)) ? &*it : nullptr;

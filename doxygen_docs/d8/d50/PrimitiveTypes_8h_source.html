53 template<typename RT, typename T>
54 static RT conversion(T const value)
55 {
56  static_assert(
57  sizeof(T) == sizeof(RT),
58  "floating point to/from integral conversion type not the same size");
59 
60  // in C it is possible to use unions to convert from one type to another (type-punning)
61  // in C++ this behavior is compiler dependent and can therefore be undefined behavior
62  // as a solution memcpy can be used to convert from one type into another type
63  RT dest{};
64  std::memcpy(&dest, &value, sizeof(value));
65  return dest;
66 }

114 template<>
115 struct PackIEEE754Dispatcher<IEC559_NOT_SUPPORTED>
116 {
117  template<typename RT, typename T, uint32_t EXP_BITS>
118  static RT packIEEE754(T const value)
119  {
120  uint32_t const BITS = (static_cast<uint32_t>(sizeof(T)) * 8U);
121  uint32_t const FRACTION_BITS = BITS - EXP_BITS - 1U;
122  if (value == static_cast<T>(0.0))
123  {
124  if (conversion<RT, T>(value) == static_cast<RT>(0.0))
125  {
126  return IEEE756Traits<T>::IEEE756_ZERO;
127  }
128 
129  return IEEE756Traits<T>::IEEE756_NEGATIVE_ZERO;
130  }
131  if (value == std::numeric_limits<T>::infinity())
132  {
133  return IEEE756Traits<T>::IEEE756_INFINITY;
134  }
135  if (value == -std::numeric_limits<T>::infinity())
136  {
137  return IEEE756Traits<T>::IEEE756_NEGATIVE_INFINITY;
138  }
139  if (value != value)
140  {
141  return IEEE756Traits<T>::IEEE756_NAN;
142  }
143 
144  T valueNormalized = value;
145  RT sign = 0U;
146  if (value < static_cast<T>(0))
147  {
148  sign = 1U;
149  valueNormalized = -valueNormalized;
150  }
151  int32_t shift = 0;
152  while ((valueNormalized >= static_cast<T>(2.0)) && (shift < IEEE756Traits<T>::MAX_EXP))
153  {
154  valueNormalized /= static_cast<T>(2.0);
155  ++shift;
156  }
157  int32_t rounds = 0;
158  while ((valueNormalized < 1.0) && (rounds < IEEE756Traits<T>::MAX_EXP))
159  {
160  valueNormalized *= static_cast<T>(2.0);
161  --shift;
162  ++rounds;
163  }
164  valueNormalized = valueNormalized - static_cast<T>(1.0);
165  RT const fraction
166  = valueNormalized * ((static_cast<RT>(1) << FRACTION_BITS) + static_cast<T>(0.5));
167  RT const exponent
168  = shift + static_cast<int32_t>(((static_cast<uint32_t>(1U) << (EXP_BITS - 1U)) - 1U));
169 
170  return (sign << (BITS - 1U)) | (exponent << FRACTION_BITS) | fraction;
171  }
172 };

187 template<>
188 struct UnpackIEEE754Dispatcher<IEC559_NOT_SUPPORTED>
189 {
190  template<typename RT, typename T, uint32_t EXP_BITS>
191  static RT unpackIEEE754(T const value)
192  {
193  uint32_t const BITS = static_cast<uint32_t>(sizeof(T) * 8U);
194  uint32_t const FRACTION_BITS = BITS - EXP_BITS - 1U;
195  if (value == IEEE756Traits<RT>::IEEE756_ZERO)
196  {
197  return static_cast<RT>(0.0);
198  }
199  if (value == IEEE756Traits<RT>::IEEE756_NEGATIVE_ZERO)
200  {
201  return -static_cast<RT>(0.0);
202  }
203  if (value == IEEE756Traits<RT>::IEEE756_INFINITY)
204  {
205  return std::numeric_limits<RT>::infinity();
206  }
207  if (value == IEEE756Traits<RT>::IEEE756_NEGATIVE_INFINITY)
208  {
209  return -std::numeric_limits<RT>::infinity();
210  }
211  T const sign = (value >> (BITS - 1U));
212  T const exponent
213  = (value & static_cast<T>(::util::meta::Bitmask<uint64_t, BITS - 1U>::value))
214  >> FRACTION_BITS;
215  T const fraction
216  = (value & static_cast<T>(::util::meta::Bitmask<uint64_t, FRACTION_BITS>::value));
217 
218  if ((::util::meta::Bitmask<uint64_t, EXP_BITS>::value == exponent) && (fraction > 0U))
219  {
220  return std::numeric_limits<RT>::quiet_NaN();
221  }
222  RT valueNormalized = static_cast<RT>(fraction);
223  int32_t shift = static_cast<int32_t>(
224  exponent - static_cast<T>(::util::meta::Bitmask<uint64_t, EXP_BITS - 1U>::value));
225 
226  valueNormalized
227  /= static_cast<T>(static_cast<T>(1U) << FRACTION_BITS) + static_cast<RT>(0.5);
228  valueNormalized += static_cast<RT>(1.0);
229 
230  while (shift < 0)
231  {
232  valueNormalized /= static_cast<RT>(2.0);
233  ++shift;
234  }
235  while (shift > 0)
236  {
237  valueNormalized *= static_cast<RT>(2.0);
238  --shift;
239  }
240  if (sign != 0U)
241  {
242  valueNormalized = -valueNormalized;
243  }
244  return valueNormalized;
245  }
246 };

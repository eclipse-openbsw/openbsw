140 template<uint8_t NUM_BUFFERS>
141 inline BufferedEventSender<NUM_BUFFERS>::BufferedEventSender(
142  INetwork& network, ::async::ContextType const ethernetContext, ITpTransceiver& tpTransceiver)
143 : ::someip::BufferedEventSender(
144  network, ethernetContext, tpTransceiver, ::etl::span<internal::EventMessage*>())
145 , _messageArray()
146 , _messageList()
147 {
148 #pragma GCC diagnostic push
149 #pragma GCC diagnostic ignored "-Wtype-limits"
150  for (uint8_t i = 0U; i < NUM_BUFFERS; ++i)
151 #pragma GCC diagnostic pop
152  {
153  internal::EventMessage& buffer = _messageArray[i];
154  _messageList.push_back(&buffer);
155  }
156  setMessages(_messageList);
